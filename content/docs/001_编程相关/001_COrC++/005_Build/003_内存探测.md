cmake 配置
```bash
-DCMAKE_C_FLAGS="-fsanitize=address -fsanitize-recover=address -fno-stack-protector -fno-omit-frame-pointer"
-DCMAKE_CXX_FLAGS="-fsanitize=address -fsanitize-recover=address -fno-stack-protector -fno-omit-frame-pointer"
-DCMAKE_EXE_LINKER_FLAGS="-fsanitize=address"
-DCMAKE_SHARED_LINKER_FLAGS="-fsanitize=address"
```
